class Solution {
    public int leastInterval(char[] tasks, int n) {

        int[] freq = new int[26];

        // Count frequency of each task
        for (char task : tasks) {
            freq[task - 'A']++;
        }

        // Find the maximum frequency
        int maxFreq = 0;

        for (int count : freq) {
            maxFreq = Math.max(maxFreq, count);
        }

        // Number of gaps created by the most frequent task
        int gaps = maxFreq - 1;

        // Each gap needs n idle/other-task positions
        int emptySlots = gaps * n;

        // Fill those slots using other tasks
        int availableTasks = tasks.length - maxFreq;

        int idles = Math.max(0, emptySlots - availableTasks);

        return tasks.length + idles;
    }
}
